⏱️ 1. Big-O Complexity ที่ต้องจำ

โครงสร้างข้อมูล Access Search Insertion Deletion

💻 2. ตัวอย่างการสร้าง Node ใน Linked List 

class Node {
  constructor(data) {
    this.data = data;
    this.next = null;
  }
}

class LinkedList {
  constructor() {
    this.head = null;
    this.size = 0;
  }

  // แทรกข้อมูลที่หัว List (O(1))
  insertAtHead(data) {
    const newNode = new Node(data);
    newNode.next = this.head;
    this.head = newNode;
    this.size++;
  }
}
